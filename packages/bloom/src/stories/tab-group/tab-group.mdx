The Tabs component offers flexibility by allowing users to navigate sections of a single page without changing the main styles and structure of the page.

> Powered by `@radix-ui/react-tabs`, this guide dives into when and how to use it for an optimal user experience.

### Features

- Can be controlled or uncontrolled.
- Supports automatic/manual activation.
- Full keyboard navigation.

## Installation

`npm install @zinnia/bloom/TabGroup`

## Anatomy

The Zinnia Tabs component extends the equivalent Radix-ui components and their props. Follow the [Radix-ui docs](https://www.radix-ui.com/primitives/docs/components/tabs) for implementation details.

```jsx
<TabGroup>
  <TabList>
    <TabTrigger />
  </TabList>
  <TabContent />
</TabGroup>
```

```jsx
import React from 'react';
import {
  TabGroup,
  TabList,
  TabTrigger,
  TabContent,
} from '@zinnia/bloom/TabGroup';
import { cardStyles } from '@zinnia/bloom/Card';
import { Icon, IconType } from '@zinnia/bloom/components';

export const TabGroupExample = () => {

  const handleTabChange = (value: unknown) => {
    console.log(value);
  };

  return (
    <TabGroup defaultValue="tab1" activationMode="manual" onValueChange={handleTabChange}>
      <TabList>
        <TabTrigger value="tab1">
          <Icon type={IconType.DOCUMENT_TEXT} /> One
        </TabTrigger>
        <TabTrigger value="tab2">
          <Icon type={IconType.DOCUMENT_TEXT} /> Two
        </TabTrigger>
        <TabTrigger value="tab3">
          <Icon type={IconType.DOCUMENT_TEXT} /> Three
        </TabTrigger>
      </TabList>
      <TabContent value="tab1" className={cardStyles.card}>
        Content for tab 1
      </TabContent>
      <TabContent value="tab2" className={cardStyles.card}>
        Content for tab 2
      </TabContent>
      <TabContent value="tab3" className={cardStyles.card}>
        Content for tab 3
      </TabContent>
    </TabGroup>
  );
};

```
